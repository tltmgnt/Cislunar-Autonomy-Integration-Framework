cmake_minimum_required(VERSION 3.16)
project(cislunar VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(cislunar src/core.cpp src/c_api.cpp)
target_include_directories(cislunar PUBLIC include)
target_compile_options(cislunar PRIVATE -Wall -Wextra -Wpedantic)
add_executable(cislunar_sim app/sim.cpp)
target_link_libraries(cislunar_sim PRIVATE cislunar)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE cislunar)
add_test(NAME core COMMAND core_tests)
add_executable(c_api_test tests/c_api_test.c)
set_target_properties(c_api_test PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(c_api_test PRIVATE cislunar)
add_test(NAME c_api COMMAND c_api_test)
add_test(NAME simulation COMMAND cislunar_sim --steps 100 --output smoke.csv)
